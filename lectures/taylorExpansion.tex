\begin{PauseHighLight}
  \begin{itemize}
  \item In much of machine learning we are trying to approximate

  \item There are many different ways to do this\pause

\begin{slide}
\section[-2]{Polynomial Approximation}

\begin{PauseHighLight}
  \begin{itemize}
  \item Suppose we wanted to approximate a function $f(x)$ around a
    point $x_0$ using a polynomial
    \begin{align*}
      f(x) \approx a_0 + a_1 (x-x_0) + a_2 (x-x_0)^2 + \ldots + a_n (x-x_0)^n\pause
    \end{align*}
  \item Taking derivatives
    \begin{align*}
      f'(x) &\approx a_1 + 2\, a_2 (x-x_0) + 3\,a_3 (x-x_0)^2 + \ldots + n\, a_n (x-x_0)^{n-1}\pause\\
      f''(x) &\approx 2\,a_2 + 6\, a_3 (x-x_0) + \ldots + n\,(n-1)\, a_n (x-x_0)^{n-2}\pauseb
    \end{align*}
  \item One way to decide on the coefficients is to set $x=x_0$ and
    assume equality
    \begin{align*}
      a_0 &= f(x_0)\pauseb, & a_1&=f'(x_0)\pauseb, & a_2 &= \frac{1}{2} f''(x_0)\pauseb, & a_i
      &= \frac{1}{i!} f^{(i)}(x_0)\pauseb
    \end{align*}
  \end{itemize}
\end{PauseHighLight}

\end{slide}

\begin{slide}
\section{Convergence}

\begin{PauseHighLight}
  \begin{itemize}
  \item As we add more and more terms our local approximation becomes
    more accurate\pause, provided the higher derivatives exist\pause
  \item Polynomials have the nice property that they only diverge at
    $\infty$\pause
  \item However, polynomials cannot capture functions that diverge for
    finite numbers, e.g.
    \begin{align*}
      &\frac{1}{x}\pause &&\frac{1}{(x-5)^2}\pauseb &&\tan(x)\pauseb 
    \end{align*}
  \item This leads to a loss of global convergence\pauseb
  \end{itemize}
\end{PauseHighLight}

\end{slide}
